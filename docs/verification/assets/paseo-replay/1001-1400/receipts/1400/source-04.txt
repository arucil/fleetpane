        next.push({

          tabId: tab.tabId,
          kind: "terminal",
          terminalId: target.terminalId,
          label: terminal?.name ?? "Terminal",
          subtitle: "Terminal",
        });
        continue;
      }
      const filePath = target.path;
      const fileName = filePath.split("/").filter(Boolean).pop() ?? filePath;
      next.push({
        key: tab.tabId,
        tabId: tab.tabId,
        kind: "file",
        filePath,
        label: fileName,
        subtitle: filePath,
      });
    }
    return next;
  }, [openTabs, sessionAgents, terminals]);
