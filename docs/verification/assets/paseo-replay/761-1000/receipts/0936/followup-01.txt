+  if (!output) {

+  const kept: string[] = [];
+
+  for (const line of output.split("\n")) {

+      const codeStr = trimmed.slice(EXIT_CODE_MARKER.length).trim();
+      const parsed = parseInt(codeStr, 10);

+  }
+
